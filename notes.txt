PM2 BASIC COMMANDs:
⤷
    chmod +x deploy.sh  
    ls -l deploy.sh   

    BACKEND: pm2 start process.config.js --env production
    FRONTEND: sh deploy.sh

    at server: chmod 755 uploads


    pm2 ls
    pm2 start dist/server.js --name=COMFYZONE
    pm2 start "npm run start:prod" --name=COMFYZONE
    pm2 start "yarn run start:prod" --name=COMFYZONE-REACT
    pm2 stop id
    pm2 start id
    pm2 restart id
    pm2 monit
    pm2 start id --watch    ||  (auto-restart on changes)
    pm2 delete <id|name>
    pm2 stop all            ||  Stop all apps
    pm2 delete all
    pm2 kill
